-- Table: public.vehicle

-- DROP TABLE IF EXISTS public.vehicle;

CREATE TABLE IF NOT EXISTS public.vehicle
(
    id bigint NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 9223372036854775807 CACHE 1 ),
    external_id uuid NOT NULL DEFAULT gen_random_uuid(),
    registration_no character varying(20) COLLATE pg_catalog."default" NOT NULL,
    created_at timestamp(6) without time zone NOT NULL DEFAULT (CURRENT_TIMESTAMP AT TIME ZONE 'UTC'::text),
    driver_id bigint,
    is_active boolean NOT NULL DEFAULT true,
    CONSTRAINT vehicle_pkey PRIMARY KEY (id),
    CONSTRAINT uq_vehicle_registration_no UNIQUE (registration_no),
    CONSTRAINT fk_vehicle_driver FOREIGN KEY (driver_id)
        REFERENCES public.driver (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.vehicle
    OWNER to postgres;

CREATE TABLE IF NOT EXISTS public.driver
(
    id bigint NOT NULL GENERATED BY DEFAULT AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 9223372036854775807 CACHE 1 ),
    external_id uuid NOT NULL DEFAULT gen_random_uuid(),
    name character varying(100) COLLATE pg_catalog."default" NOT NULL,
    phone character varying(20) COLLATE pg_catalog."default" NOT NULL,
    preferred_language_id smallint NOT NULL,
    language_source character varying(10) COLLATE pg_catalog."default" NOT NULL DEFAULT 'default'::character varying,
    do_not_call boolean NOT NULL DEFAULT false,
    created_at timestamp(6) without time zone NOT NULL DEFAULT (CURRENT_TIMESTAMP AT TIME ZONE 'UTC'::text),
    updated_at timestamp(6) without time zone NOT NULL DEFAULT (CURRENT_TIMESTAMP AT TIME ZONE 'UTC'::text),
    CONSTRAINT driver_pkey PRIMARY KEY (id),
    CONSTRAINT uq_driver_external_id UNIQUE (external_id),
    CONSTRAINT uq_driver_phone UNIQUE (phone),
    CONSTRAINT fk_driver_language FOREIGN KEY (preferred_language_id)
        REFERENCES public.language (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT ck_driver_language_source CHECK (language_source::text = ANY (ARRAY['default'::character varying, 'manual'::character varying, 'detected'::character varying]::text[])),
    CONSTRAINT ck_driver_phone_e164 CHECK (phone::text ~ '^[+][1-9][0-9]{7,14}$'::text)
)

TABLESPACE pg_default;

ALTER TABLE IF EXISTS public.driver
    OWNER to postgres;
